cmake_minimum_required(VERSION 3.20)
project(shopping_sessions_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
set(SHOPPING_ARROW_ROOT "" CACHE PATH "Arrow headers and libraries for native Parquet output")

add_library(session_engine src/engine.cpp)
target_include_directories(session_engine PUBLIC include)
target_link_libraries(session_engine PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
if(SHOPPING_ARROW_ROOT)
    find_library(SHOPPING_ARROW_LIBRARY arrow PATHS "${SHOPPING_ARROW_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    find_library(SHOPPING_PARQUET_LIBRARY parquet PATHS "${SHOPPING_ARROW_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_compile_definitions(session_engine PUBLIC SHOPPING_NATIVE_PARQUET)
    target_include_directories(session_engine SYSTEM PRIVATE "${SHOPPING_ARROW_ROOT}/include")
    target_link_libraries(session_engine PRIVATE "${SHOPPING_PARQUET_LIBRARY}" "${SHOPPING_ARROW_LIBRARY}")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(session_engine PUBLIC -march=x86-64-v3)
    endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(session_engine PRIVATE -Wall -Wextra -Wconversion)
endif()

add_executable(cpp_engine src/main.cpp)
target_link_libraries(cpp_engine PRIVATE session_engine)
if(SHOPPING_ARROW_ROOT)
    install(TARGETS cpp_engine RUNTIME DESTINATION .)
    set_target_properties(cpp_engine PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(cpp_engine PRIVATE -static-libstdc++ -static-libgcc)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE session_engine)
    add_test(NAME engine_tests COMMAND engine_tests)
endif()
